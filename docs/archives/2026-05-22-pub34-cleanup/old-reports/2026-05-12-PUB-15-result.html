⏰ useAlertTimer.ts — 임계값 알림 훅

자기장이 줄어드는 중(isShrinking=true)일 때만 30/20/10초 알림을 트리거.
동일 임계값 중복 방지 + 자기장 재시작 감지 시 alertedRef 자동 초기화.

// 재시작 감지: 이전보다 초가 증가하면 리셋
if (prevSecondsRef.current !== null
    && remainingSeconds > prevSecondsRef.current) {
  alertedRef.current.clear();
}

// 임계값 도달 시 1회만 발송
if (thresholds.includes(remainingSeconds)
    && !alertedRef.current.has(remainingSeconds)) {
  alertedRef.current.add(remainingSeconds);
  onAlert(remainingSeconds);
}
useCallback useRef 중복방지 재시작감지
